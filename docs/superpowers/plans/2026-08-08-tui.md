# FinAgent TUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the rich Console CLI loop with a Textual TUI — single-column chat layout with streaming agent replies, tool-call progress lines, message queueing, and Esc interrupt.

**Architecture:** One file `finagent/tui.py` replaces `cli.py`. A Textual `App` subclass composes four widgets: Header (`Static`), ChatView (`VerticalScroll` holding message `Static` widgets), Input (`Input`), StatusBar (`Static`). Agent streaming runs in a `@work` async worker that calls `agent.astream(stream_mode=["messages","updates"])` — "messages" for token streaming, "updates" for tool-call progress. A message queue allows typing during agent processing; queued messages merge and flush when the turn ends.

**Tech Stack:** Textual >= 0.40, rich (Markdown rendering), LangGraph astream, pytest + pytest-asyncio + Textual Pilot for testing.

```python
"""FinAgent TUI — Textual terminal interface."""
import uuid

from textual.app import App, ComposeResult
from textual.containers import VerticalScroll
from textual.widgets import Input, Static

from finagent.agent import create_agent, reset_checkpoint
from finagent.config import load_env
from finagent.report import generate_report

HELP_TEXT = """\
可用命令:
  /report   生成财报点评报告（基于当前对话历史）
  /clear    清空对话记忆（换公司分析时用）
  /help     显示此帮助
  /quit     退出
"""

_COMMANDS = {"/quit", "/help", "/clear", "/report"}


def parse_command(text: str) -> tuple[str, str]:
    """Parse user input into (command_name, payload).

    Returns ("message", text) for non-command input.
    For slash commands, returns (command_name_without_slash, "").
    """
    stripped = text.strip()
    lower = stripped.lower()
    if lower in _COMMANDS:
        return lower[1:], ""
    return "message", stripped


class FinAgentApp(App):
    CSS = """
    Screen { layout: vertical; }
    #header { height: 1; background: $boost; padding: 0 1; }
    #chat-view { height: 1fr; padding: 0 1; }
    #input { height: 1; }
    #status { height: 1; background: $boost; color: $text-muted; padding: 0 1; }
    .message-user { color: $primary; }
    .message-tool { color: $text-muted; }
    .message-error { color: $error; }
    .message-queued { color: $text-disabled; text-style: italic; }
    """

    def __init__(self):
        super().__init__()
        load_env()
        self.thread_id = str(uuid.uuid4())
        self.agent = create_agent()
        self._queue: list[str] = []
        self._queued_widgets: list[Static] = []
        self._streaming_worker = None

    def compose(self) -> ComposeResult:
        yield Static("FinAgent — A股财报点评助手", id="header")
        yield VerticalScroll(id="chat-view")
        yield Input(placeholder="输入股票代码 + 报告期，如 002415 2024Q3", id="input")
        yield Static("就绪", id="status")

    def on_mount(self) -> None:
        self._add_message("输入股票代码 + 报告期开始分析。输入 /help 查看命令。")

    def on_input_submitted(self, event: Input.Submitted) -> None:
        text = event.value.strip()
        event.input.value = ""
        if not text:
            return
        cmd, payload = parse_command(text)
        if cmd == "quit":
            self.exit()
        elif cmd == "help":
            self._add_message(HELP_TEXT)
        # message, clear, report: wired in Tasks 2 and 4

    def _add_message(self, content, classes: str = "") -> Static:
        """Mount a message widget into the chat view, scroll to bottom."""
        widget = Static(content, classes=classes)
        self.query_one("#chat-view").mount(widget)
        self.query_one("#chat-view").scroll_end(animate=False)
        return widget

    def _set_status(self, text: str) -> None:
        self.query_one("#status").update(text)


def main():
    app = FinAgentApp()
    app.run()
```

```python
    def _submit_message(self, text: str) -> None:
        """Send a message to the agent, or enqueue if agent is busy."""
        if self._streaming_worker is not None and self._streaming_worker.is_running:
            self._queue.append(text)
            widget = self._add_message(f"[排队中] {text}", classes="message-queued")
            self._queued_widgets.append(widget)
        else:
            self._add_message(f"> {text}", classes="message-user")
            self._streaming_worker = self._start_stream(text)

    @work
    async def _start_stream(self, user_input: str) -> None:
        """Stream agent response: tool-call progress + token-by-token reply."""
        self._set_status("思考中...")
        reply_widget = self._add_message("")
        buffer = ""
        tool_names: dict[str, str] = {}
        try:
            async for mode, data in self.agent.astream(
                {"messages": [{"role": "user", "content": user_input}]},
                config={"configurable": {"thread_id": self.thread_id}},
                stream_mode=["messages", "updates"],
            ):
                if get_current_worker().is_cancelled():
                    break
                if mode == "messages":
                    chunk, _metadata = data
                    content = getattr(chunk, "content", "")
                    has_tool_calls = bool(getattr(chunk, "tool_call_chunks", None))
                    if content and not has_tool_calls:
                        buffer += content
                        reply_widget.update(buffer)
                elif mode == "updates":
                    for node, state in data.items():
                        for msg in state.get("messages", []):
                            tool_calls = getattr(msg, "tool_calls", None)
                            if tool_calls:
                                for tc in tool_calls:
                                    name = tc.get("name", "tool")
                                    tc_id = tc.get("id", "")
                                    tool_names[tc_id] = name
                                    self._add_message(f"🔧 {name} ⏳", classes="message-tool")
                            elif hasattr(msg, "tool_call_id"):
                                name = tool_names.get(msg.tool_call_id, getattr(msg, "name", "tool"))
                                self._add_message(f"🔧 {name} ✓", classes="message-tool")
            if buffer:
                reply_widget.update(Markdown(buffer))
        except Exception as e:
            self._add_message(f"出错: {e}", classes="message-error")
        finally:
            self._set_status("就绪")
            self._streaming_worker = None
```

    def action_interrupt(self) -> None:
        """Esc handler: cancel current streaming worker."""
        if self._streaming_worker is not None and self._streaming_worker.is_running:
            self._streaming_worker.cancel()
            self._add_message("[已中断]", classes="message-queued")

    def _do_clear(self) -> None:
        """Clear conversation memory and chat view."""
        reset_checkpoint()
        self.thread_id = str(uuid.uuid4())
        self.agent = create_agent()
        self._queue.clear()
        self._queued_widgets.clear()

    def _do_report(self) -> None:
        """Generate report from conversation history, show summary inline."""
        state = self.agent.get_state(
            config={"configurable": {"thread_id": self.thread_id}}
        )
        messages = state.values.get("messages", []) if state and state.values else []
        if not messages:
            self._add_message("还没有对话内容，先聊几句再生成报告。", classes="message-error")
            return
        self._set_status("正在生成报告...")
        try:
            msg_dicts = [{"role": m.type, "content": m.content} for m in messages]
            filepath, content = generate_report(msg_dicts)
            # extract title (first markdown heading)
            title = "未命名报告"

## Notes for Implementer

- **Streaming event format:** `astream(stream_mode=["messages", "updates"])` yields `(mode, data)` tuples. "messages" mode gives `(chunk, metadata)`; "updates" mode gives `{node: state_dict}`.
- **Async workers:** `@work` (without `thread=True`) runs on the Textual event loop — can directly call widget methods like `mount()`, `update()`, `scroll_end()`. Do NOT use `call_from_thread` (that's only for `thread=True` workers).
- **Worker cancellation:** `self._start_stream(text)` returns a `Worker` object. Store it. Call `.cancel()` to interrupt. Inside the worker, check `get_current_worker().is_cancelled()` in the loop.
- **Pilot test timing:** After triggering a worker, call `await pilot.app.workers.wait_for_complete()` + `await pilot.pause()` to let it finish before asserting.
- **agent.get_state()** is used for `/report` to pull conversation history from the checkpointer. Returns an object with `.values` dict containing `messages` list.
